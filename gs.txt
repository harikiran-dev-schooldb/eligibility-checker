/************************************************************
 WHATSAPP MESSAGE TEMPLATES
 (USED ONLY ON STATUS CHANGES – NOT ON SUBMIT)
*************************************************************/
function waApplicationIssued(student, parent, enquiryNo) {
  return `
📄 *Application Form Issued*
*Enquiry No:* ${enquiryNo}

Dear Parent (${parent}),

The application form for your child has been issued.

👦 *Student:* ${student}

Kindly submit the completed form at the school office.

Kotak Salesian School`;
}

function waApplicationNotIssued(student, parent, enquiryNo) {
  return `
📄 *Application Form Not Issued*
*Enquiry No:* ${enquiryNo}

Dear Parent (${parent}),

The application form has not been issued yet.

👦 *Student:* ${student}

Please collect the application form from the school office.

Kotak Salesian School`;
}

function waEntranceResult(student, parent, result, enquiryNo) {
  if (result === "PASS") {
    return `
✅ *Entrance Test Result – PASS*
*Enquiry No:* ${enquiryNo}

Congratulations! ${student} has passed the entrance test.

Kotak Salesian School`;
  }

  if (result === "FAIL") {
    return `
❌ *Entrance Test Result – FAIL*
*Enquiry No:* ${enquiryNo}

Thank you for your interest.

Kotak Salesian School`;
  }

  return "";
}

function waInterviewResult(student, parent, status, enquiryNo) {
  return `
🎤 *Interview Status*
*Enquiry No:* ${enquiryNo}

👦 *Student:* ${student}
📊 *Status:* ${status}

Kotak Salesian School`;
}

function waFinalAdmission(student, parent, status, enquiryNo) {
  if (status === "YES") {
    return `
🎉 *Admission Confirmed*
*Enquiry No:* ${enquiryNo}

Dear Parent (${parent}),

${student} has been admitted.

Please complete fee formalities at the earliest.

Welcome to Kotak Salesian School!`;
  }

  return `
❌ *Admission Not Approved*
*Enquiry No:* ${enquiryNo}

Thank you for your interest.

Kotak Salesian School`;
}


/************************************************************
 WHATSAPP LINK BUILDER (SHOWS ONLY “Link”)
*************************************************************/
function makeWhatsAppLink(mobile, message) {
  mobile = String(mobile).replace(/\D/g, "").slice(-10);
  const url =
    `https://web.whatsapp.com/send?phone=91${mobile}&text=${encodeURIComponent(message)}`;
  return `=HYPERLINK("${url}", "Link")`;
}


/************************************************************
 SAVE ENQUIRY — WEB APP POST
 (NO WHATSAPP MESSAGE GENERATED HERE)
*************************************************************/
function doPost(e) {
  const data = JSON.parse(e.postData.contents);

  const sheet = SpreadsheetApp
    .openById("1he10Az5om3tJjvorLOasxbSZpWQjY3S-WSEPMaDFJSU")
    .getSheetByName("Admissions 2026-27");

  const row = sheet.getLastRow() + 1;

  sheet.getRange(row, 1, 1, 14).setValues([[
    enquiryNo,
    data.student,
    data.admClass,
    data.parent,
    data.mobile,
    data.dob,

    data.eligibleClass,
    Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd-MM-yyyy"),
    "",                 // ✅ WhatsApp link empty on submit

    "NOT STARTED",
    "PENDING",
    "NO"
  ]]);

  sheet.getRange(row, 1, 1, 14).setBackground("#bbdefb");

  return ContentService.createTextOutput(
    JSON.stringify({ status: "success", enquiryNo })
  )
    .setMimeType(ContentService.MimeType.JSON)
    .setHeader("Access-Control-Allow-Origin", "*");
}


/************************************************************
 ON EDIT — WHATSAPP + COLOR CONTROL
*************************************************************/
function onEdit(e) {
  const sheet = e.source.getActiveSheet();
  if (sheet.getName() !== "Admissions 2026-27") return;

  const row = e.range.getRow();
  const col = e.range.getColumn();
  if (row < 4) return;

  const student   = sheet.getRange(row, 2).getValue();
  const parent    = sheet.getRange(row, 4).getValue();
  const mobile    = sheet.getRange(row, 5).getValue();
  const enquiryNo = sheet.getRange(row, 1).getValue();

  let message = "";
  let color = "";

  if (col === 11) {
    message = e.value === "YES"
      ? waApplicationIssued(student, parent, enquiryNo)
      : waApplicationNotIssued(student, parent, enquiryNo);
    if (e.value === "YES") color = "#fff9c4";
  }

  if (col === 12) {
    message = waEntranceResult(student, parent, e.value, enquiryNo);
    color = e.value === "PASS" ? "#c8e6c9" : "#ffcdd2";
  }

  if (col === 13) {
    message = waInterviewResult(student, parent, e.value, enquiryNo);
    color = "#d1c4e9";
  }

  if (col === 14) {
    message = waFinalAdmission(student, parent, e.value, enquiryNo);
    if (e.value === "YES") color = "#a5d6a7";
  }

  if (!message) return;

  sheet.getRange(row, 10).setValue(makeWhatsAppLink(mobile, message));
  sheet.getRange(row, 9).setValue(
    Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd-MM-yyyy")
  );

  if (color) sheet.getRange(row, 1, 1, 14).setBackground(color);
}

/**************************************************
  GET (JSONP - for dashboard)
**************************************************/
function doGet(e) {
  const action = e.parameter.action;
  const callback = e.parameter.callback || "callback";
  let result = {};

  if (action === "getAdmissions") {
    result = getAdmissionsData();
  }

  return ContentService
    .createTextOutput(`${callback}(${JSON.stringify(result)})`)
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}


/**************************************************
  POST (save enquiry OR update stage)
**************************************************/
function doPost(e) {
  const out = HtmlService.createHtmlOutput("");
  out.addMetaTag("Access-Control-Allow-Origin", "*")
     .addMetaTag("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
     .addMetaTag("Access-Control-Allow-Headers", "Content-Type");

  try {
    const data = JSON.parse(e.postData.contents);

    // 1️⃣ UPDATE STAGE (Dashboard)
    if (data.action === "updateStage") {
      updateStageValue(data.enquiryNo, data.field, data.value);
      out.setContent(JSON.stringify({ status: "updated" }));
      return out;
    }

    // 2️⃣ SAVE NEW ENQUIRY (Eligibility Page)
    if (data.parent) {
      const enquiryNo = generateEnquiryNumber();
      saveEnquiry(enquiryNo, data);

      out.setContent(JSON.stringify({ status: "saved", enquiryNo }));
      return out;
    }

    out.setContent(JSON.stringify({ status: "unknown action" }));
    return out;

  } catch (err) {
    out.setContent(JSON.stringify({ status: "error", message: err.toString() }));
    return out;
  }
}


/**************************************************
  SHEET FUNCTIONS
**************************************************/
function getSheet() {
  const ss = SpreadsheetApp.openById("1he10Az5om3tJjvorLOasxbSZpWQjY3S-WSEPMaDFJSU");
  return ss.getSheetByName("Admissions 2026-27");
}

function getAdmissionsData() {
  const sheet = getSheet();
  const rows = sheet.getRange(4, 1, sheet.getLastRow() - 3, 14).getValues();

  return rows.map(r => ({
    enquiryNo: r[0],
    student: r[1],
    admClass: r[2],
    parent: r[3],
    mobile: r[4],
    dob: r[5],
    age: r[6],
    eligible: r[7],
    date: r[8],
    application: r[10],
    entrance: r[11],
    interview: r[12],
    finalAdmission: r[13]
  }));
}

function generateEnquiryNumber() {
  const sheet = getSheet();
  const lastRow = sheet.getLastRow();

  if (lastRow < 4) return "ENQ-2026-0001";

  const last = sheet.getRange(lastRow, 1).getValue();
  const num = parseInt(last.split("-")[2], 10) + 1;

  return `ENQ-2026-${String(num).padStart(4, "0")}`;
}

function saveEnquiry(enq, data) {
  const sheet = getSheet();

  sheet.appendRow([
    enq,
    data.student,
    data.admClass,
    data.parent,
    data.mobile,
    data.dob,
    data.age,
    data.eligibleClass,
    Utilities.formatDate(new Date(), "Asia/Kolkata", "dd-MM-yyyy"),
    "",
    "NO",
    "NOT STARTED",
    "PENDING",
    "NO"
  ]);
}

function updateStageValue(enq, field, value) {
  const sheet = getSheet();
  const data = sheet.getRange(4, 1, sheet.getLastRow() - 3, 14).getValues();

  const colMap = {
    application: 11,
    entrance: 12,
    interview: 13,
    finalAdmission: 14
  };

  const col = colMap[field];
  if (!col) return;

  for (let i = 0; i < data.length; i++) {
    if (data[i][0] === enq) {
      sheet.getRange(i + 4, col).setValue(value);
      return;
    }
  }
}
